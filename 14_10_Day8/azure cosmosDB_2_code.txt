CREATE DATABASE hotel_assessment;
--1--
SELECT * FROM guests;

--2--
SELECT guest_name, city, membership_level
FROM guests;

--3--
SELECT *
FROM rooms
WHERE price_per_night > 5000;

--4--
SELECT *
FROM rooms
WHERE status = 'Available';

--5--
SELECT *
FROM guests
WHERE membership_level = 'Gold';

--6--
SELECT *
FROM bookings
WHERE booking_status = 'Confirmed';

--7--
SELECT *
FROM bookings
ORDER BY check_in_date;

--8--
SELECT *
FROM rooms
ORDER BY price_per_night DESC;

--9--
SELECT *
FROM rooms
WHERE price_per_night BETWEEN 4000 AND 8000;

--10--
SELECT *
FROM guests
WHERE city IN ('Hyderabad', 'Pune', 'Mumbai');
--11--
SELECT COUNT(*) AS total_rooms
FROM rooms;

--12--
SELECT AVG(price_per_night) AS avg_room_price
FROM rooms;

--13--
SELECT MIN(price_per_night) AS lowest_price
FROM rooms;

--14--
SELECT MIN(price_per_night) AS lowest_price
FROM rooms;

--15--
SELECT room_type,
       COUNT(*) AS total_rooms
FROM rooms
GROUP BY room_type;

--16--
SELECT room_type,
       AVG(price_per_night) AS avg_price
FROM rooms
GROUP BY room_type;

--17--
SELECT membership_level,
       COUNT(*) AS total_guests
FROM guests
GROUP BY membership_level;

--18--
SELECT booking_status,
       COUNT(*) AS total_bookings
FROM bookings
GROUP BY booking_status;

--19--
SELECT SUM(service_price) AS total_service_value
FROM services;

--20--
SELECT AVG(service_price) AS avg_service_price
FROM services;

--21--
SELECT b.booking_id,
       g.guest_name,
       g.city,
       b.check_in_date,
       b.check_out_date,
       b.booking_status
FROM guests g
JOIN bookings b
ON g.guest_id = b.guest_id;

--22--
SELECT b.booking_id,
       r.room_type,
       r.price_per_night,
       b.check_in_date,
       b.check_out_date
FROM bookings b
JOIN rooms r
ON b.room_id = r.room_id;

--23--
SELECT b.booking_id,
       r.room_type,
       r.price_per_night,
       b.check_in_date,
       b.check_out_date
FROM bookings b
JOIN rooms r
ON b.room_id = r.room_id;

--24--
SELECT b.booking_id,
       s.service_name,
       bs.quantity,
       s.service_price
FROM bookings b
JOIN booking_services bs
ON b.booking_id = bs.booking_id
JOIN services s
ON bs.service_id = s.service_id;

--25--
SELECT b.booking_id,
       s.service_name,
       bs.quantity,
       s.service_price,
       (bs.quantity * s.service_price) AS service_cost
FROM bookings b
JOIN booking_services bs
ON b.booking_id = bs.booking_id
JOIN services s
ON bs.service_id = s.service_id;

--26--
SELECT g.guest_name,
       b.booking_id,
       r.room_type,
       s.service_name,
       bs.quantity,
       b.booking_status
FROM guests g
JOIN bookings b
ON g.guest_id = b.guest_id
JOIN rooms r
ON b.room_id = r.room_id
JOIN booking_services bs
ON b.booking_id = bs.booking_id
JOIN services s
ON bs.service_id = s.service_id;

--27--
SELECT g.guest_name,
       b.booking_id,
       b.booking_status
FROM guests g
LEFT JOIN bookings b
ON g.guest_id = b.guest_id;

--28--
SELECT g.guest_id,
       g.guest_name,
       g.city
FROM guests g
LEFT JOIN bookings b
ON g.guest_id = b.guest_id
WHERE b.booking_id IS NULL;

--29--
SELECT g.guest_name,
       COUNT(b.booking_id) AS total_bookings
FROM guests g
LEFT JOIN bookings b
ON g.guest_id = b.guest_id
GROUP BY g.guest_name
ORDER BY total_bookings DESC;
